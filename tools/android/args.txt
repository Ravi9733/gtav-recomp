# Stable Android baseline for Snapdragon 8+ Gen 1 / Adreno 730.
# Prefer the game's native 30 fps limiter and synchronized presentation over uncapped
# rendering, which increases heat and causes larger frame-time swings on mobile.
--headless=true
--log_level=info
--gtav_native=on
--gtav_fps_limit=30
--gpu_skip_rendering=true
--gpu_skip_rendering_lag_ms=6
--vsync=true
--gtav_native_vsync=true
--vulkan_log_debug_messages=false
--window_presenter=false
# Keep the existing bounded texture cache; do not lower image quality to chase FPS.
--gtav_native_texture_budget_mb=384

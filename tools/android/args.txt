# Sustained-play profile for Android Adreno devices (including Snapdragon 8+ Gen 1).
# 30 fps pacing is preferred over uncapped bursts that cause heat and frame-time spikes.
--headless=true
--log_level=info
--gtav_native=on
--gtav_fps_limit=30
--gpu_skip_rendering=true
--gpu_skip_rendering_lag_ms=6
--vsync=true
--gtav_native_vsync=true
--vulkan_log_debug_messages=false
--window_presenter=false
# Keep the texture cache bounded to leave memory for the game, Android and GPU driver.
--gtav_native_texture_budget_mb=384
